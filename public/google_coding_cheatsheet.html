1. 并查集 Union-Find

信号:「分组 / 连通性 / 传递性 / 朋友圈 / 动态加边」。三件套:parent 数组 + find(路径压缩) + union(计组数)。每操作近似 O(1)(α 反阿克曼)。

#547 Number of Provinces — 模板

union 所有相连城市,维护组数计数器,返回最终组数。
class Solution:
    def findCircleNum(self, isConnected):
        n = len(isConnected)
        parent = list(range(n))
        self.count = n                       # 组数
        def find(x):
            while parent[x] != x:
                parent[x] = parent[parent[x]]   # 路径压缩
                x = parent[x]
            return x
        def union(x, y):
            rx, ry = find(x), find(y)
            if rx != ry:
                parent[rx] = ry
                self.count -= 1              # 合并成功,组数-1
        for i in range(n):
            for j in range(i+1, n):
                if isConnected[i][j] == 1:
                    union(i, j)
        return self.count
时间 O(n²·α) ≈ O(n²) · 空间 O(n)

#1101 Earliest Moment Everyone Become Friends

同模板,多两步:按 timestamp 排序 + 组数变 1 时返回当前时刻。
def earliestAcq(self, logs, n):
    logs.sort()                              # ① 按时间排序
    parent = list(range(n)); self.count = n
    # find / union 同 #547 ...
    for t, a, b in logs:
        union(a, b)
        if self.count == 1:                  # ③ 全连通
            return t
    return -1
时间 O(E log E) · 空间 O(n)

#737 Sentence Similarity II

同义词对 union 成组;逐位置检查 w1==w2 或 find 同根。传递性 → 并查集。
def areSentencesSimilarTwo(self, s1, s2, pairs):
    if len(s1) != len(s2): return False
    parent = {}
    def find(x):
        parent.setdefault(x, x)
        if parent[x] != x: parent[x] = find(parent[x])
        return parent[x]
    for a, b in pairs: parent[find(a)] = find(b)
    return all(w1==w2 or find(w1)==find(w2) for w1,w2 in zip(s1,s2))
时间 O((P+N)·α) · 空间 O(词数)
组数套路: union 成功 count-=1。#547 返回最终组数;#1101 监控变 1;#323 连通分量数。

2. 区间 / 扫描线 Intervals & Sweep Line

信号:「区间合并 / 重叠 / 最大并发 / 空隙」。套路:排序 + 扫描(维护 prev_end 或堆或差分)。判重叠公式:s1 < e2 and s2 < e1。

#56 Merge Intervals / #759 / #3169 — 合并 & 找空隙

排序 + prev_end 扫描:重叠则延伸,否则记录新段/空隙。
# #3169 数空闲天:合并忙碌区间,缝即空闲
def countDays(self, days, meetings):
    meetings.sort(key=lambda x: x[0])
    free, prev_end = 0, 0
    for s, e in meetings:
        if s > prev_end + 1:
            free += s - prev_end - 1         # 空隙
        prev_end = max(prev_end, e)          # 注意 max(内嵌区间)
    if days > prev_end: free += days - prev_end
    return free
时间 O(n log n) · 空间 O(1)

#253 Meeting Rooms II — 最大并发(最少房间)

按 start 排序,最小堆存进行中会议的 end;堆顶≤当前start则复用。堆大小=房间数。
import heapq
def minMeetingRooms(self, intervals):
    intervals.sort(key=lambda x: x[0])
    heap = []
    for s, e in intervals:
        if heap and heap[0] <= s:
            heapq.heappop(heap)              # 复用
        heapq.heappush(heap, e)
    return len(heap)
时间 O(n log n) · 空间 O(n)

#2402 Meeting Rooms III — 双堆 + 等待

idle 堆存空房号,busy 堆存(end,房号);没空房则延后等最早结束的。
def mostBooked(self, n, meetings):
    meetings.sort()
    idle = list(range(n)); heapq.heapify(idle)
    busy = []; count = [0]*n
    for s, e in meetings:
        while busy and busy[0][0] <= s:
            _, r = heapq.heappop(busy); heapq.heappush(idle, r)
        if idle:
            r = heapq.heappop(idle); heapq.heappush(busy, (e, r))
        else:
            end, r = heapq.heappop(busy)
            heapq.heappush(busy, (end + (e-s), r))
        count[r] += 1
    return count.index(max(count))
时间 O(m log m + m log n) · 空间 O(n)

3. 栈 Stack

信号:「嵌套 / 括号 / 表达式 / 后进先出」。栈顶永远是 [-1]。

#20 Valid Parentheses

左括号入栈,右括号匹配栈顶,最后查栈空。
def isValid(self, s):
    stack = []; pairs = {')':'(', ']':'[', '}':'{'}
    for ch in s:
        if ch in pairs:
            if not stack or stack[-1] != pairs[ch]: return False
            stack.pop()
        else: stack.append(ch)
    return not stack
时间 O(n) · 空间 O(n)

#394 Decode String — 栈处理嵌套

遇 [ 压(当前串,倍数),遇 ] 弹出拼接 prev + k*cur。
def decodeString(self, s):
    stack = []; cur = ""; num = 0
    for ch in s:
        if ch.isdigit(): num = num*10 + int(ch)
        elif ch == '[': stack.append((cur, num)); cur, num = "", 0
        elif ch == ']': prev, k = stack.pop(); cur = prev + k*cur
        else: cur += ch
    return cur
时间 O(输出长) · 空间 O(嵌套深度)

#224 / #227 Basic Calculator — 表达式求值

#224 括号压栈现场;#227 优先级:*/ 立即算、+- 延迟求和。
# #227: + - * / 无括号
def calculate(self, s):
    stack = []; num = 0; op = '+'
    for i, ch in enumerate(s):
        if ch.isdigit(): num = num*10 + int(ch)
        if ch in '+-*/' or i == len(s)-1:
            if op == '+': stack.append(num)
            elif op == '-': stack.append(-num)
            elif op == '*': stack.append(stack.pop()*num)
            else: stack.append(int(stack.pop()/num))
            op, num = ch, 0
    return sum(stack)
时间 O(n) · 空间 O(n)

4. 单调栈 Monotonic Stack 新

信号:「下一个更大/更小元素」「每个元素左右第一个比它大/小的」「柱状图面积」「接雨水」。核心:维护一个单调(递增或递减)的栈,新元素入栈前弹出破坏单调性的,弹出时即可结算答案。每个元素进出栈各一次 → O(n)。

#739 Daily Temperatures — 下一个更暖的天

维护递减栈(存下标);新温度更高时,弹出的元素「等到了更暖的天」。
def dailyTemperatures(self, temps):
    res = [0]*len(temps)
    stack = []                               # 存下标,温度递减
    for i, t in enumerate(temps):
        while stack and temps[stack[-1]] < t:
            j = stack.pop()
            res[j] = i - j                   # 等了 i-j 天
        stack.append(i)
    return res
时间 O(n) · 空间 O(n)

#496 Next Greater Element I

递减栈;弹出时记录「下一个更大」到哈希表。
def nextGreaterElement(self, nums1, nums2):
    nxt = {}; stack = []
    for x in nums2:
        while stack and stack[-1] < x:
            nxt[stack.pop()] = x
        stack.append(x)
    return [nxt.get(x, -1) for x in nums1]
时间 O(n+m) · 空间 O(n)

#84 Largest Rectangle in Histogram — 经典难题

维护递增栈;弹出时以该高度为矮边算矩形宽度=右边界-左边界-1。末尾加哨兵 0 清空栈。
def largestRectangleArea(self, heights):
    stack = []; max_area = 0
    heights = heights + [0]                  # 哨兵,逼出剩余
    for i, h in enumerate(heights):
        while stack and heights[stack[-1]] > h:
            height = heights[stack.pop()]
            width = i if not stack else i - stack[-1] - 1
            max_area = max(max_area, height * width)
        stack.append(i)
    return max_area
时间 O(n) · 空间 O(n)

#42 Trapping Rain Water — 接雨水(双指针版)

每个位置存水 = min(左最高,右最高) - 自己高度。双指针 O(1) 空间最优(也可单调栈)。
def trap(self, height):
    left, right = 0, len(height)-1
    lmax = rmax = water = 0
    while left < right:
        if height[left] < height[right]:
            lmax = max(lmax, height[left])
            water += lmax - height[left]; left += 1
        else:
            rmax = max(rmax, height[right])
            water += rmax - height[right]; right -= 1
    return water
时间 O(n) · 空间 O(1)
口诀:「找右边/左边第一个更大」→ 递减栈;「找更小 / 算面积矮边」→ 递增栈。弹出的瞬间就是结算答案的时机。
家族:#739 · #496/#503 · #84 · #42 · #316 去重 · #901 股价跨度。

5. 堆 / Top-K Heap

信号:「第 K 大/小」「Top K」「数据流中位数」「反复取极值+合并」。取最大用负数模拟 max-heap。

#215 Kth Largest — Quick Select

partition + 只递归含目标的一半。第 k 大 = 升序第 (n-k) 位。随机 pivot 防最坏。
import random
def findKthLargest(self, nums, k):
    target = len(nums) - k
    lo, hi = 0, len(nums)-1
    while lo <= hi:
        p = self.partition(nums, lo, hi)
        if p == target: return nums[p]
        elif p < target: lo = p+1
        else: hi = p-1
def partition(self, nums, lo, hi):
    r = random.randint(lo, hi); nums[r], nums[hi] = nums[hi], nums[r]
    pivot = nums[hi]; i = lo
    for j in range(lo, hi):
        if nums[j] < pivot: nums[i], nums[j] = nums[j], nums[i]; i += 1
    nums[i], nums[hi] = nums[hi], nums[i]; return i
时间 O(n) 平均 / O(n²) 最坏 · 空间 O(1)

#347 Top K Frequent — 桶排序 O(n)

Counter 统计 + 按频率分桶(索引=频率),从高频往低扫凑 k 个。
def topKFrequent(self, nums, k):
    cnt = Counter(nums); n = len(nums)
    buckets = [[] for _ in range(n+1)]
    for num, f in cnt.items(): buckets[f].append(num)
    res = []
    for i in range(n, 0, -1):
        for num in buckets[i]:
            res.append(num)
            if len(res) == k: return res
时间 O(n) · 空间 O(n)

#1167 Connect Sticks — Huffman 贪心

每次合并最短两根(早合并的被反复累加)。最小堆。
def connectSticks(self, sticks):
    heapq.heapify(sticks); total = 0
    while len(sticks) > 1:
        a = heapq.heappop(sticks); b = heapq.heappop(sticks)
        total += a + b; heapq.heappush(sticks, a + b)
    return total
时间 O(n log n) · 空间 O(1)
三种 Top-K:Quick Select O(n) 一次性 · 堆 O(n log k) 数据流 · 桶排序 O(n)。数据流中位数:双堆(#295/#480/#1825)。

6. DFS / 回溯 / 岛屿

两类区别:
· 连通块/数岛:seen 永久标记,不回溯(否则重复计数);flag「全称性」初始 True。
· 路径搜索:seen add→remove 回溯(格子可被不同路径复用);flag「存在性」初始 False;返回值要接住。

岛屿三兄弟(#200 / #1254 / #1905)

题陆地值flag翻 False 条件
#200 数量'1'(字符串)无—
#1254 闭合0(反!)is_closed碰边界
#1905 子岛1(两网格)is_subgrid1 对应为 0
# #1254 闭合岛(0=陆地,1=水)
def closedIsland(self, grid):
    m, n = len(grid), len(grid[0]); seen = set(); is_closed = True
    def dfs(i, j):
        nonlocal is_closed
        if i<0 or i>=m or j<0 or j>=n or grid[i][j]==1 or (i,j) in seen: return
        seen.add((i,j))                      # 永久标记,不回溯
        if i==0 or i==m-1 or j==0 or j==n-1: is_closed = False
        for dx, dy in [(0,1),(0,-1),(1,0),(-1,0)]: dfs(i+dx, j+dy)
    count = 0
    for i in range(m):
        for j in range(n):
            if grid[i][j]==0 and (i,j) not in seen:
                is_closed = True; dfs(i,j)
                if is_closed: count += 1
    return count
时间 O(m·n) · 空间 O(m·n)

#79 Word Search — 路径回溯

每格启动 DFS,seen add/remove 回溯,found flag 初始 False,返回值接住。
def exist(self, grid, word):
    m, n = len(grid), len(grid[0]); seen = set()
    def dfs(i, j, cnt):
        if cnt == len(word): return True      # 先判!在越界前
        if i<0 or i>=m or j<0 or j>=n or (i,j) in seen or grid[i][j]!=word[cnt]:
            return False
        seen.add((i,j)); found = False
        for dx, dy in [(0,1),(0,-1),(1,0),(-1,0)]:
            if dfs(i+dx, j+dy, cnt+1): found = True; break
        seen.remove((i,j))                   # 回溯
        return found
    return any(dfs(i,j,0) for i in range(m) for j in range(n))
时间 O(m·n·3^L) · 空间 O(L)

#489 Robot Room Cleaner — puzzle 回溯

相对坐标 (0,0);方向数组顺时针配 turnRight;go_back() 物理复位;循环末尾 turnRight 同步朝向。
def cleanRoom(self, robot):
    dirs = [(-1,0),(0,1),(1,0),(0,-1)]; visited = set()
    def go_back():
        robot.turnRight(); robot.turnRight(); robot.move()
        robot.turnRight(); robot.turnRight()
    def bt(r, c, d):
        visited.add((r,c)); robot.clean()
        for i in range(4):
            nd = (d+i) % 4; dx, dy = dirs[nd]
            nr, nc = r+dx, c+dy
            if (nr,nc) not in visited and robot.move():
                bt(nr, nc, nd); go_back()
            robot.turnRight()                # ★ 同步物理朝向
    bt(0, 0, 0)
时间 O(N-M) · 空间 O(N-M)

#743 Network Delay — Dijkstra 模板

def networkDelayTime(self, times, n, k):
    graph = collections.defaultdict(list)
    for u, v, w in times: graph[u].append((v, w))
    dist = {}; heap = [(0, k)]
    while heap:
        d, node = heapq.heappop(heap)
        if node in dist: continue     # 第一次弹出即最短
        dist[node] = d
        for nei, w in graph[node]:
            if nei not in dist: heapq.heappush(heap, (d+w, nei))
    return max(dist.values()) if len(dist)==n else -1
时间 O(E log V) · 空间 O(V+E)

#1631 Path Min Effort — 最小化最大(瓶颈)

COMBINE = max(已有瓶颈, 相邻差);min-heap。
def minimumEffortPath(self, heights):
    m, n = len(heights), len(heights[0])
    effort = [[float('inf')]*n for _ in range(m)]; effort[0][0]=0
    heap = [(0,0,0)]
    while heap:
        e, x, y = heapq.heappop(heap)
        if x==m-1 and y==n-1: return e
        if e > effort[x][y]: continue
        for dx, dy in [(0,1),(0,-1),(1,0),(-1,0)]:
            nx, ny = x+dx, y+dy
            if 0<=nx<m and 0<=ny<n:
                ne = max(e, abs(heights[nx][ny]-heights[x][y]))
                if ne < effort[nx][ny]:
                    effort[nx][ny] = ne; heapq.heappush(heap, (ne, nx, ny))
时间 O(mn log mn) · 空间 O(mn)

#2115 Recipes — 拓扑排序(Kahn)

supplies 入度0先入队;recipe 入度=配料数;料齐则解锁。
def findAllRecipes(self, recipes, ingredients, supplies):
    graph = defaultdict(list); indeg = defaultdict(int)
    for i, r in enumerate(recipes):
        for ing in ingredients[i]:
            graph[ing].append(r); indeg[r] += 1
    queue = deque(supplies); res = []
    while queue:
        item = queue.popleft()
        for r in graph[item]:
            indeg[r] -= 1
            if indeg[r] == 0: res.append(r); queue.append(r)
    return res
时间 O(V+E) · 空间 O(V+E)

8. Trie 字典树

信号:「前缀匹配 / 多词搜索 / 自动补全 / 词典」。核心:嵌套 dict,沿字符走,结尾标记。加频率→补全,加 DFS→网格,加路径→文件系统。

#208 Implement Trie

class Trie:
    def __init__(self): self.root = {}
    def insert(self, word):
        node = self.root
        for ch in word: node = node.setdefault(ch, {})
        node['#'] = True
    def search(self, word):
        node = self._find(word); return node is not None and '#' in node
    def startsWith(self, p): return self._find(p) is not None
    def _find(self, s):
        node = self.root
        for ch in s:
            if ch not in node: return None
            node = node[ch]
        return node
insert/search/startsWith O(L) · 空间 O(总字符数)

#212 Word Search II — Trie + DFS

所有词建 Trie('#'存整词),只扫一遍 board,沿 Trie 走 + 剪枝。
def findWords(self, board, words):
    trie = {}
    for w in words:
        node = trie
        for ch in w: node = node.setdefault(ch, {})
        node['#'] = w
    m, n = len(board), len(board[0]); res = []
    def dfs(i, j, node):
        ch = board[i][j]
        if ch not in node: return
        nxt = node[ch]
        if '#' in nxt: res.append(nxt['#']); del nxt['#']
        board[i][j] = '#'
        for di, dj in [(0,1),(0,-1),(1,0),(-1,0)]:
            ni, nj = i+di, j+dj
            if 0<=ni<m and 0<=nj<n and board[ni][nj]!='#': dfs(ni, nj, nxt)
        board[i][j] = ch
    for i in range(m):
        for j in range(n): dfs(i, j, trie)
    return res

#450 Delete Node in BST

def deleteNode(self, root, key):
    if not root: return None
    if key < root.val: root.left = self.deleteNode(root.left, key)
    elif key > root.val: root.right = self.deleteNode(root.right, key)
    else:
        if not root.left: return root.right
        if not root.right: return root.left
        succ = root.right
        while succ.left: succ = succ.left
        root.val = succ.val
        root.right = self.deleteNode(root.right, succ.val)
    return root

#1522 Diameter N-Ary / #543 树直径

def diameter(self, root):
    self.ans = 0
    def depth(node):
        top1 = top2 = 0
        for ch in node.children:
            d = depth(ch) + 1
            if d > top1: top1, top2 = d, top1
            elif d > top2: top2 = d
        self.ans = max(self.ans, top1 + top2)
        return top1
    depth(root); return self.ans

10. 二分 Binary Search

#704 Binary Search — 模板

def search(self, nums, target):
    lo, hi = 0, len(nums) - 1            # 闭区间
    while lo <= hi:                       # <=
        mid = (lo + hi) // 2
        if nums[mid] == target: return mid
        elif nums[mid] < target: lo = mid + 1
        else: hi = mid - 1
    return -1

from collections import OrderedDict
class LRUCache:
    def __init__(self, capacity):
        self.cap = capacity; self.cache = OrderedDict()
    def get(self, key):

        self.cache.move_to_end(key); return self.cache[key]

#307 Range Sum Query Mutable — 模板题

线段树:叶子存值,内部存合并;update沿路径向上;query爬树取边角块。
class NumArray:                          # 线段树(迭代)
    def __init__(self, nums):
        self.n = len(nums); self.tree = [0]*(2*self.n)
        for i in range(self.n): self.tree[self.n+i] = nums[i]
        for i in range(self.n-1, 0, -1):
            self.tree[i] = self.tree[2*i] + self.tree[2*i+1]
    def update(self, index, val):
        i = index + self.n; self.tree[i] = val
        while i > 1:
            i //= 2; self.tree[i] = self.tree[2*i] + self.tree[2*i+1]
    def sumRange(self, left, right):
        l, r = left+self.n, right+1+self.n   # 含两端→半开
        res = 0
        while l < r:
            if l & 1: res += self.tree[l]; l += 1
            if r & 1: r -= 1; res += self.tree[r]
            l //= 2; r //= 2
        return res
build O(n) · update/query O(log n) · 空间 O(n)

BIT 树状数组(求和更简洁)

class BIT:
    def __init__(self, n): self.tree = [0]*(n+1)
    def update(self, i, delta):           # 1-indexed
        while i < len(self.tree):
            self.tree[i] += delta; i += i & (-i)   # lowbit
    def query(self, i):                   # 前缀和[1,i]
        s = 0
        while i > 0: s += self.tree[i]; i -= i & (-i)
        return s
update/query O(log n) · 空间 O(n)

#315 Count of Smaller After Self — BIT 应用

离散化 + 从右往左,query(rank-1)=右边比它小的个数,再 update(rank)。
def countSmaller(self, nums):
    rank = {v:i+1 for i,v in enumerate(sorted(set(nums)))}
    n = len(rank); tree = [0]*(n+1); res = []
    def upd(i):
        while i <= n: tree[i]+=1; i += i&(-i)
    def qry(i):
        s=0
        while i>0: s+=tree[i]; i-=i&(-i)
        return s
    for x in reversed(nums):
        r = rank[x]; res.append(qry(r-1)); upd(r)
    return res[::-1]
时间 O(n log n) · 空间 O(n)

def minimumSwap(self, s1, s2):
    xy = yx = 0
    for a, b in zip(s1, s2):
        if a=='x' and b=='y': xy += 1
        elif a=='y' and b=='x': yx += 1
